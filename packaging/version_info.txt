# Windows version resource for the packaged build.
#
# An executable with no CompanyName, ProductName or FileVersion is unusual for
# real software and common for malware, and Defender's machine-learning models
# weigh that. Filling it in is both honest metadata and one less reason to be
# flagged. Keep the four version numbers in step with the release tag.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(2, 8, 3, 0),
    prodvers=(2, 8, 3, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [
          StringStruct('CompanyName', 'AquAxs'),
          StringStruct('FileDescription', 'My Jarvis - personal AI assistant'),
          StringStruct('FileVersion', '2.8.3.0'),
          StringStruct('InternalName', 'jarvis'),
          StringStruct('LegalCopyright', 'MIT Licence'),
          StringStruct('OriginalFilename', 'jarvis.exe'),
          StringStruct('ProductName', 'My Jarvis'),
          StringStruct('ProductVersion', '2.8.3.0'),
        ]
      )
    ]),
    VarFileInfo([VarStruct('Translation', [0x0409, 1200])])
  ]
)
